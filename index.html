h+=`<button style="width:100%;margin-bottom:6px" onclick="F.rows.push({...F.rows[F.rows.length-1]});rRows()">+ Serie</button>`}

$('#rows').innerHTML=h}

function rSess(){chips($('#sg'),S.session.groups,g=>{const a=S.session.groups,i=a.indexOf(g);i<0?a.push(g):a.splice(i,1);save();rSess()});

$('#tplSel').innerHTML='<option value="">— Scegli scheda —</option>'+S.templates.map(t=>`<option value="${t.id}">${esc(t.name)}</option>`).join('');

$('#sess').innerHTML=ex.length?ex.map((e,i)=>`<div class="ex"><div class="row" style="margin:0"><b>${esc(e.name)} <span class="mu">${e.type==='cardio'?'· Cardio':'· Forza'}${e.prog?' · prog.':''}</span></b><button class="x fx" onclick="S.session.exercises.splice(${i},1);save();rSess()">✕</button></div>${e.sets.map((s,j)=>`<span class="set">S${j+1} ${fmt(e.type,s)}</span>`).join('')}</div>`).join(''):'<span class="mu">Nessun esercizio aggiunto.</span>'}

function saveW(){const s=S.session;if(!s.exercises.length){alert('Aggiungi almeno un esercizio.');return}

const n=prompt('Nome scheda:',s.groups.join(' + ')||'Scheda');if(!n)return;

S.templates.push({id:uid(),name:n.trim(),groups:[...s.groups],exercises:copy(s.exercises)});save();rSess();rTpls()}

function loadTpl(){const t=S.templates.find(x=>x.id===$('#tplSel').value);if(!t)return;

S.session={groups:[...t.groups],exercises:copy(t.exercises).map(e=>({...e,id:uid()}))};save();rSess()}

function rTpls(){$('#tpls').innerHTML=S.templates.length?S.templates.map(t=>`<div class="ex"><b>${esc(t.name)}</b><div class="mu">${t.groups.map(esc).join(' + ')||'—'} · ${t.exercises.length} esercizi</div>

$('#cal').innerHTML=h;rDet()}

$('#det').innerHTML=L.length?L.map(l=>`<div class="ex"><div class="row" style="margin:0"><div class="chips" style="margin:0">${l.groups.map(g=>`<span class="chip on">${esc(g)}</span>`).join('')||'<span class="mu">Nessun gruppo</span>'}</div><button class="x fx" onclick="delLog('${l.id}')">🗑</button></div>

${l.exercises.map(e=>`<div style="margin-top:8px"><b>${esc(e.name)}</b> <span class="mu">${e.type==='cardio'?'Cardio':'Forza'}</span><br>${e.sets.map((s,j)=>`<span class="set">S${j+1} ${fmt(e.type,s)}</span>`).join('')}</div>`).join('')}</div>`).join(''):'<span class="mu">Nessun allenamento in questa data.</span>'}

r.readAsText(f);e.target.value=''};


/* Init */

$('#wDate').value=dk(new Date());C.sel=dk(new Date());rRows();rSess();
